<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /

  # Serve real files and directories (like admin/ or uploads/) directly
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  # Send everything else to the API router
  RewriteRule ^ router/index.php [QSA,L]

</IfModule>
